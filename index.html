样张一 · 数据清洗工具（Python）

输入：含 BOM、混合换行、非法日期、重复行的 CSV
def clean_csv(path, *, date_col="created_at", drop_dupes=True):
    """清洗 CSV：去 BOM、统一换行、规范化日期、去重。"""
    with open(path, encoding="utf-8-sig", newline="") as f:
        rows = list(csv.DictReader(f))

    cleaned, bad = [], []
    for i, r in enumerate(rows, 2):          # 从第 2 行开始（含表头）
        r = {k.strip(): (v or "").strip() for k, v in r.items()}
        if not r.get("id"):
            bad.append((i, "缺少 id")); continue
        r[date_col] = normalize_date(r.get(date_col))   # 返回 ISO8601 或 None
        if r[date_col] is None:
            bad.append((i, f"非法日期: {rows[i-2].get(date_col)!r}")); continue
        cleaned.append(r)

    if drop_dupes:
        seen, uniq = set(), []
        for r in cleaned:
            if r["id"] in seen: continue
            seen.add(r["id"]); uniq.append(r)
        cleaned = uniq

    return cleaned, bad       # bad: [(行号, 原因)]，便于回溯原始数据
